2. POSTMAN LOGIN REQUESTS AND TOKEN SCRIPTS
===========================================

POST {{baseUrl}}/auth/login

Admin body:

{
  "email": "{{adminEmail}}",
  "password": "{{password}}"

pm.test("Admin login succeeded", function () {
  pm.response.to.have.status(201);

4. RUN A NEW AGENT-TO-COORDINATOR FLOW
======================================

STEP 1 — ADMIN FUNDS THE COORDINATOR
------------------------------------

POST {{baseUrl}}/wallets/{{coordinatorWalletId}}/fund
Bearer {{adminToken}}

{
  "amount": 50000
}


STEP 2 — AGENT REPORTS LOW CASH
-------------------------------

POST {{baseUrl}}/balance-snapshots
Bearer {{agentToken}}

{
  "agentId": "{{agentId}}",
  "cashBalance": 5000,
  "eMoneyBalance": 100000,
  "recordedAt": "2026-07-30T13:00:00.000Z"
}

Tests script:

pm.environment.set("newBeforeSnapshotId", pm.response.json().id);


STEP 3 — AGENT REQUESTS MONEY
-----------------------------

POST {{baseUrl}}/liquidity-requests
Bearer {{agentToken}}

{
  "requesterId": "{{agentId}}",
  "requesterType": "AGENT",
  "liquidityType": "PHYSICAL_CASH",
  "providerId": "{{providerId}}",
  "areaId": "{{areaId}}",
  "amount": 50000,
  "urgency": "HIGH",
  "allowPartial": false,
  "expiresAt": "2027-12-31T23:59:59.000Z",
  "notes": "Postman live cash request"
}

Tests script:

pm.test("Request created", function () {
  pm.response.to.have.status(201);
});
pm.environment.set("newRequestId", pm.response.json().id);


STEP 4 — COORDINATOR LISTS OPEN REQUESTS
----------------------------------------

GET {{baseUrl}}/liquidity-requests?providerId={{providerId}}&status=open
Bearer {{coordinatorToken}}


STEP 5 — COORDINATOR SUBMITS AN OFFER
-------------------------------------

POST {{baseUrl}}/liquidity-offers
Bearer {{coordinatorToken}}

{
  "requestId": "{{newRequestId}}",
  "coordinatorId": "{{coordinatorId}}",
  "availableAmount": 50000,
  "etaMinutes": 20,
  "note": "Coordinator can fulfil this request"
}

Tests script:

pm.environment.set("newOfferId", pm.response.json().id);


STEP 6 — AGENT ACCEPTS THE OFFER
--------------------------------

POST {{baseUrl}}/liquidity-offers/{{newOfferId}}/accept
Bearer {{agentToken}}

{
  "assignedAmount": 50000
}

Tests script:

pm.environment.set("newAssignmentId", pm.response.json().id);


STEP 7 — COORDINATOR TRANSFERS THE MONEY
----------------------------------------

POST {{baseUrl}}/liquidity-transfers
Bearer {{coordinatorToken}}

Header:

Idempotency-Key: postman-live-transfer-{{$timestamp}}

Body:

{
  "fromWalletId": "{{coordinatorWalletId}}",
  "toWalletId": "{{agentWalletId}}",
  "amount": 50000,
  "transferType": "COORDINATOR_SUPPLY",
  "requestId": "{{newRequestId}}",
  "assignmentId": "{{newAssignmentId}}"
}

Tests script:

pm.test("Transfer succeeded", function () {
  pm.response.to.have.status(201);
  pm.expect(pm.response.json().status).to.eql("success");
});
pm.environment.set("newTransferId", pm.response.json().id);


STEP 8 — AGENT REPORTS THE NEW BALANCE

POST {{baseUrl}}/balance-snapshots
Bearer {{agentToken}}

{

7. EXPECTED BUSINESS-RULE ERRORS
================================

400:
  Invalid DTO fields
  Missing Idempotency-Key
  Insufficient points
  Insufficient wallet balance
  Updating a non-open request
  Withdrawing a non-submitted offer

401:
  Missing, invalid, or expired JWT

403:
  Correct JWT but wrong role

404:
  Unknown resource ID

409:
  Duplicate email, phone, or tenant code
  Reused idempotency key with a different body
  Accepting an already accepted offer
  Deleting a wallet with a non-zero balance
